1744:     geo.translate(-run / 2, 0, -s.width / 2);

1746:     m.rotation.y = Math.atan2(-s.runZ, s.runX);
1747:     m.position.set(s.x, s.baseY, s.z);
1748:     m.castShadow = true; m.receiveShadow = true;
1749:     scene.add(m);
