261: // Lay a hollow pyramid of barrels: row 0 sits on the floor, each row above
262: // nests in the groove of the one below (hex packing). Barrels lie on their
263: // side, so the pyramid spreads perpendicular to the barrel axis.
264: export function barrelPyramid({ x, z, rows, alongX = true, radius = 0.55, length = 1.3 }) {
265:   const out = [];
266:   const dx = radius * 2;
267:   const dy = radius * Math.sqrt(3);
268:   for (let k = 0; k < rows; k++) {
269:     const n = rows - k;
270:     const rowW = (n - 1) * dx;
271:     for (let i = 0; i < n; i++) {
272:       const o = -rowW / 2 + i * dx;
273:       const y = radius + k * dy;
274:       out.push({ x: alongX ? x : x + o, y, z: alongX ? z + o : z, alongX, length, radius });
275:     }
276:   }
277:   return out;
278: }
