247: // ===== Corn field =====
248: // A corn plant is a lot of little parts â€” a tapered stem with leaf nodes, six
249: // big alternating leaves, a spray of tassel at the crown and one or two husked
250: // ears with a yellow cob poking out. Drawing that per stalk with a dozen meshes
251: // each would be thousands of draw calls, so the whole plant is baked into ONE
252: // vertex-coloured geometry and stamped across the field with InstancedMesh.
253: // Colour lives in the vertices (stem / leaf / husk / cob / tassel), which lets a
254: // single double-sided material draw it all, and the field stays three draws
255: // wide. A handful of variants (different leaf angles and ear placement) breaks

258: // One leaf: a long tapered blade lying along +X and arching up towards +Y,
259: // widest around the middle and pointed at both ends.
260: function makeCornLeafGeometry() {
261:   const segs = 6;
262:   const pos = [];
263:   const idx = [];
264:   for (let i = 0; i <= segs; i++) {
265:     const t = i / segs;
266:     const y = 0.42 * Math.sin(Math.PI * t);
267:     const w = 0.5 * 0.5 * Math.pow(Math.sin(Math.PI * t), 0.65);
268:     pos.push(t, y, -w, t, y, w);
269:   }
270:   for (let s = 0; s < segs; s++) {
271:     const a = s * 2;
272:     idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
273:   }
274:   const g = new THREE.BufferGeometry();
275:   g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
276:   g.setIndex(idx);
277:   g.computeVertexNormals();
278:   return g;

289: function cornPart(acc, geo, matrix, color) {
290:   const p = geo.attributes.position.array;
291:   const nrm = geo.attributes.normal.array;
292:   const nm = new THREE.Matrix3().getNormalMatrix(matrix);
293:   const base = acc.pos.length / 3;
294:   const v = new THREE.Vector3();
295:   for (let i = 0; i < p.length; i += 3) {
296:     v.set(p[i], p[i + 1], p[i + 2]).applyMatrix4(matrix);
297:     acc.pos.push(v.x, v.y, v.z);
298:     v.set(nrm[i], nrm[i + 1], nrm[i + 2]).applyMatrix3(nm).normalize();
299:     acc.nrm.push(v.x, v.y, v.z);
300:     acc.col.push(color.r, color.g, color.b);
301:   }
302:   const gi = geo.index.array;
303:   for (let i = 0; i < gi.length; i++) acc.idx.push(base + gi[i]);
304: }

312: // Bake one whole corn plant into a single geometry. `seed` nudges the leaf and
313: // ear placement so a few variants can share the field without looking cloned.
314: function buildCornStalkGeometry(seed) {
315:   let s = 1 + seed * 17.3;
316:   const rnd = () => { s = (s * 9301 + 49297) % 233280; return (s / 233280) * 2 - 1; };
317:   const H = CORN.stalkH;
318:   const acc = { pos: [], nrm: [], idx: [], col: [] };
319: 
320:   // Stem: a tapered stalk with a few swollen leaf nodes up its length.
321:   cornPart(acc, new THREE.CylinderGeometry(0.1, 0.19, H, 6), cornPlace(0, H / 2, 0, 0, 0, 0), CORN_STALK_COL);
322:   for (let k = 0; k < 3; k++) {
323:     cornPart(acc, new THREE.CylinderGeometry(0.15, 0.15, 0.1, 6), cornPlace(0, H * (0.3 + k * 0.22), 0, 0, 0, 0), CORN_STALK_COL);
324:   }
325: 
326:   // Six big leaves, each twisted to a new angle around the stalk and alternated
327:   // between an upright and a droopy arch.
328:   for (let i = 0; i < 6; i++) {
329:     const y = H * (0.15 + i * 0.115) + rnd() * 0.06;
330:     const yaw = i * 2.399963 + rnd() * 0.4;
331:     const droop = i % 2 ? 0.72 : 0.42;
332:     const len = 1.35 + rnd() * 0.45;
333:     cornPart(acc, CORN_LEAF_GEO, cornPlace(0, y, 0, -0.12, yaw, droop + rnd() * 0.12, len, len * 0.9, len), CORN_LEAF_COL);
334:   }
335: 
336:   // Ears: husk blades fanning out of the stem, a yellow cob poking free of them,
337:   // and a little brown silk tuft at the tip.
338:   for (let e = 0; e < 2; e++) {
339:     const ey = H * (0.5 + e * 0.14) + rnd() * 0.05;
340:     const yaw = e * 3.1 + 0.6 + rnd() * 0.5;
341:     const pitch = 0.85 + rnd() * 0.2;
342:     const dx = Math.sin(pitch) * Math.sin(yaw);
343:     const dz = Math.sin(pitch) * Math.cos(yaw);
344:     for (let h = 0; h < 3; h++) {
345:       cornPart(acc, CORN_LEAF_GEO, cornPlace(dx * 0.08, ey, dz * 0.08, -0.05, yaw + h * 2.0944, 0.8, 0.55, 0.6, 0.55), CORN_HUSK_COL);
346:     }
347:     cornPart(acc, new THREE.CylinderGeometry(0.075, 0.06, 0.42, 8), cornPlace(dx * 0.2, ey + Math.cos(pitch) * 0.2, dz * 0.2, pitch, yaw, 0), CORN_COB_COL);
348:     cornPart(acc, new THREE.ConeGeometry(0.03, 0.16, 4), cornPlace(dx * 0.44, ey + Math.cos(pitch) * 0.44, dz * 0.44, pitch, yaw, 0), CORN_HUSK_COL);
349:   }
350: 
351:   // Tassel: a spray of thin spikes above the crown.
352:   for (let t = 0; t < 5; t++) {
353:     const a = t * 1.2566;
354:     const lean = 0.35 + (t % 2) * 0.28;
355:     cornPart(acc, new THREE.ConeGeometry(0.035, 0.6, 4), cornPlace(Math.sin(a) * 0.05, H + 0.22, Math.cos(a) * 0.05, lean, a, 0), CORN_TASSEL_COL);
356:   }
357: 
358:   const g = new THREE.BufferGeometry();
359:   g.setAttribute('position', new THREE.Float32BufferAttribute(acc.pos, 3));
360:   g.setAttribute('normal', new THREE.Float32BufferAttribute(acc.nrm, 3));
361:   g.setAttribute('color', new THREE.Float32BufferAttribute(acc.col, 3));
362:   g.setIndex(acc.idx);
363:   return g;
364: }

366: function buildCorn(scene) {
367:   const pts = cornGrid();
368:   const n = pts.length;
369: 
370:   // Split the field across a few geometry variants so the rows aren't identical.
371:   const VARIANTS = 3;
372:   const counts = new Array(VARIANTS).fill(0);
373:   for (let i = 0; i < n; i++) counts[i % VARIANTS]++;
374:   const meshes = [];
375:   for (let v = 0; v < VARIANTS; v++) {
376:     const im = new THREE.InstancedMesh(buildCornStalkGeometry(v + 1), M.cornStalk, counts[v]);
377:     im.frustumCulled = false;
378:     im.castShadow = true;
379:     scene.add(im);
380:     meshes.push(im);
381:   }
382: 
383:   const knocked = new Uint8Array(n);
384:   const done = new Uint8Array(n);
385:   const angle = new Float32Array(n);
386:   const dirX = new Float32Array(n);
387:   const dirZ = new Float32Array(n);
388:   const meshOf = new Uint8Array(n);
389:   const slot = new Int32Array(n);
390:   const yaw = new Float32Array(n);
391:   const scale = new Float32Array(n);
392:   const cursor = new Array(VARIANTS).fill(0);
393:   for (let i = 0; i < n; i++) {
394:     const v = i % 3;
395:     meshOf[i] = v;
396:     slot[i] = cursor[v]++;
397:     yaw[i] = (i * 2.399963) % (Math.PI * 2);
398:     scale[i] = 0.9 + ((i * 37) % 21) / 100;
399:   }
400: 
401:   const writeStalk = (i, a) => {
402:     const p = pts[i];
403:     _q.identity();
404:     if (a > 0.0001) {
405:       _axis.set(dirZ[i], 0, -dirX[i]);
406:       if (_axis.lengthSq() < 1e-6) _axis.set(1, 0, 0);
407:       _axis.normalize();
408:       _q.setFromAxisAngle(_axis, a);
409:     }
410:     _q.multiply(_qy.setFromAxisAngle(_up, yaw[i]));
411:     _v.set(p.x, CORN_GROUND, p.z);
412:     _m.compose(_v, _q, _sc.setScalar(scale[i]));
413:     meshes[meshOf[i]].setMatrixAt(slot[i], _m);
414:   };
415:   for (let i = 0; i < n; i++) writeStalk(i, 0);
416:   for (const im of meshes) im.instanceMatrix.needsUpdate = true;
417: 
418:   return {
419:     count: n,
420:     pts, knocked, done, angle, dirX, dirZ, meshes,
421:     // Bend one stalk over in the direction the car is travelling.
422:     knock(i, headingX, headingZ, onCob) {
423:       if (knocked[i]) return;
424:       knocked[i] = 1;
425:       dirX[i] = headingX; dirZ[i] = headingZ;
426:       angle[i] = 0;
427:       onCob(pts[i]);
428:     },
429:     // Advance the fall animation.
430:     step(dt) {
431:       let any = false;
432:       for (let i = 0; i < n; i++) {
433:         if (!knocked[i] || done[i]) continue;
434:         angle[i] += (Math.PI / 2) * (dt / CORN.fallTime);
435:         if (angle[i] >= Math.PI / 2) { angle[i] = Math.PI / 2; done[i] = 1; }
436:         writeStalk(i, angle[i]);
437:         any = true;
438:       }
439:       if (any) for (const im of meshes) im.instanceMatrix.needsUpdate = true;
440:     },
441:     reset() {
442:       for (let i = 0; i < n; i++) { knocked[i] = 0; done[i] = 0; angle[i] = 0; }
443:       for (let i = 0; i < n; i++) writeStalk(i, 0);
444:       for (const im of meshes) im.instanceMatrix.needsUpdate = true;
445:     },
446:   };
447: }

563:   // A strut between two points: a thin box oriented along the segment.
564:   const strut = (ax, ay, az, bx, by, bz, t, mat) => {
565:     const dx = bx - ax, dy = by - ay, dz = bz - az;
566:     const len = Math.hypot(dx, dy, dz) || 0.001;
567:     const m = new THREE.Mesh(new THREE.BoxGeometry(t, len, t), mat);
568:     m.position.set((ax + bx) / 2, (ay + by) / 2, (az + bz) / 2);
569:     m.quaternion.setFromUnitVectors(_up, _v.set(dx / len, dy / len, dz / len));
570:     m.castShadow = true;
571:     scene.add(m);
572:     return m;
573:   };
574:   // Point on a leaning leg at height y.
575:   const legPt = (sx, sz, y) => {
576:     const s = legBase + (legTop - legBase) * (y / legH);
577:     return [cx + sx * s, y, cz + sz * s];

867: // Aim a ballistic arc from (sx,sy,sz) to (tx,ty,tz) at speed S.
868: // Returns a velocity vector, or null when the target is out of practical range.
869: function aimArc(sx, sy, sz, tx, ty, tz, speed) {
870:   const dx = tx - sx, dz = tz - sz;
871:   const dist = Math.hypot(dx, dz);
872:   if (dist < 0.3) return null;
873:   const t = dist / speed;
874:   const nx = dx / dist, nz = dz / dist;
875:   const vy = (ty - sy) / t + 0.5 * APPLE_GRAVITY * t;
876:   return { vx: nx * speed, vy, vz: nz * speed };
877: }

919: // Barrels are stacked in pyramids but each one is a free body: bump the pile
920: // and the barrels burst apart and roll away like logs. A barrel lies on its
921: // side, so it only rolls broadside (see stepBarrel) â€” the long axis stays put.
922: // `state` is 'stacked' (still on the pile), 'falling' (dropped off the pile),
923: // 'rolling', or 'rest'. Every barrel above a knocked one comes down with it.
924: function buildBarrels(scene) {
925:   const barrels = [];
926:   for (const stack of BARREL_STACKS) {
927:     for (const b of barrelPyramid(stack)) {
928:       const g = makeBarrel(true, b.radius * 2, b.radius);
929:       const axis = b.alongX ? 0 : Math.PI / 2;
930:       g.position.set(b.x, b.y, b.z);
931:       g.rotation.set(0, axis, 0);
932:       scene.add(g);
933:       barrels.push({
934:         group: g,
935:         radius: b.radius,
936:         axis,
937:         x: b.x, y: b.y, z: b.z,
938:         vx: 0, vz: 0, vy: 0, roll: 0,
939:         state: 'stacked',
940:         home: { x: b.x, y: b.y, z: b.z },
941:         above: [],
942:       });
943:     }
944:   }
945:   // Which barrels rest on each barrel: anything a row or more higher that is
946:   // within reach of its groove. Knock a barrel out and everything on top of it
947:   // is released together.
948:   for (const a of barrels) {
949:     for (const c of barrels) {
950:       if (c === a) continue;
951:       if (c.y > a.y + 0.5 && Math.hypot(c.x - a.x, c.z - a.z) < a.radius * 1.7) a.above.push(c);
952:     }
953:   }
954:   return barrels;
955: }

957: // Knock one barrel into the field: roll it broadside (the hit projected onto the
958: // plane perpendicular to its long axis, so even an end-on bump rolls it aside)
959: // and release every barrel that was stacked on top of it.
960: function knockBarrel(b, nx, nz, power) {
961:   const ax = Math.cos(b.axis), az = -Math.sin(b.axis);
962:   const along = nx * ax + nz * az;
963:   let px = nx - along * ax;
964:   let pz = nz - along * az;
965:   let pl = Math.hypot(px, pz);
966:   if (pl < 1e-3) { px = -az; pz = ax; pl = 1; }
967:   b.vx = (px / pl) * power;
968:   b.vz = (pz / pl) * power;
969:   b.state = 'rolling';
970:   for (const o of b.above) {
971:     if (o.state !== 'stacked') continue;
972:     o.state = 'falling';
973:     const a = Math.random() * Math.PI * 2;
974:     o.vx = Math.cos(a) * 0.9;
975:     o.vz = Math.sin(a) * 0.9;
976:     o.vy = 0;
977:   }
978: }
