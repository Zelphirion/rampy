70: // ===== Spiral ramp generator =====
71: // A chain of short straight ramp defs whose ends meet exactly, forming a helix
72: // the car can drive up. Returns defs shaped like the underground ramp defs:
73: // { x, z, runX, runZ, len, width, height, baseY, boost }.
74: export function spiralRampSegments({ cx, cz, startAngle = 0, r = 3, y0 = 0, y1 = 12, turns = 2.2, segsPerTurn = 18, width = 3 } = {}) {
75:   const total = Math.max(1, Math.round(turns * segsPerTurn));
76:   const segs = [];
77:   let px = cx + r * Math.cos(startAngle);
78:   let pz = cz + r * Math.sin(startAngle);
79:   let py = y0;
80:   for (let k = 1; k <= total; k++) {
81:     const t = k / total;
82:     const phi = startAngle + t * turns * Math.PI * 2;
83:     const nx = cx + r * Math.cos(phi);
84:     const nz = cz + r * Math.sin(phi);
85:     const dx = nx - px, dz = nz - pz;
86:     const len = Math.hypot(dx, dz) || 0.001;
87:     const y = y0 + (y1 - y0) * t;
88:     const runX = dx / len, runZ = dz / len;
89:     segs.push({
90:       x: (px + nx) / 2, z: (pz + nz) / 2,
91:       runX, runZ, len, width,
92:       height: y - py, baseY: py, boost: 0,
93:     });
94:     px = nx; pz = nz; py = y;
95:   }
96:   return segs;
97: }

119: export function siloColliders() {
120:   const n = 20;
121:   const out = [];
122:   for (let i = 0; i < n; i++) {
123:     const a = (i / n) * Math.PI * 2;
124:     let d = a - SILO.gapAngle;
125:     while (d > Math.PI) d -= Math.PI * 2;
126:     while (d < -Math.PI) d += Math.PI * 2;
127:     if (Math.abs(d) < 0.5) continue;           // the entrance gap
128:     out.push({
129:       x: SILO.x + SILO.r * Math.cos(a),
130:       z: SILO.z + SILO.r * Math.sin(a),
131:       halfW: 0.75, halfD: 0.75,
132:       h: SILO.spiralTop, top: SILO.spiralTop,
133:     });
134:   }
135:   return out;
136: }

138: // ===== Corn grid =====
139: export function cornGrid(rect = CORNFIELD_RECT, spacing = CORN.spacing) {
140:   const pts = [];
141:   const nx = Math.floor((rect.x1 - rect.x0) / spacing);
142:   const nz = Math.floor((rect.z1 - rect.z0) / spacing);
143:   const ox = (rect.x0 + rect.x1) / 2 - (nx * spacing) / 2;
144:   const oz = (rect.z0 + rect.z1) / 2 - (nz * spacing) / 2;
145:   for (let i = 0; i < nx; i++) {
146:     for (let j = 0; j < nz; j++) {
147:       const x = ox + i * spacing + Math.sin(i * 1.7 + j * 0.9) * 0.35;
148:       const z = oz + j * spacing + Math.cos(i * 0.8 - j * 1.3) * 0.35;
149:       pts.push({ x, z });
150:     }
151:   }
152:   return pts;
153: }

155: // Does a car at (carX, carZ) moving at `speed` bend this stalk over?
156: export function shouldKnockStalk(carX, carZ, speed, stalk, radius = CORN.knockR) {
157:   if (Math.abs(speed) < CORN.minSpeed) return false;
158:   const dx = carX - stalk.x, dz = carZ - stalk.z;
159:   return dx * dx + dz * dz < radius * radius;
160: }

162: // ===== Pumpkin rolling =====
163: // Mutates a plain {x,z,vx,vz,spin,radius}. Returns the pre-friction speed so
164: // the caller knows whether it is still rolling. Pure enough to unit-test.
165: export function stepPumpkin(p, dt, drag = 1.8) {
166:   const sp = Math.hypot(p.vx, p.vz);
167:   if (sp < 0.02) { p.vx = 0; p.vz = 0; return 0; }
168:   p.x += p.vx * dt;
169:   p.z += p.vz * dt;
170:   p.spin += (sp * dt) / Math.max(0.2, p.radius || 1);
171:   const f = Math.max(0, 1 - drag * dt);
172:   p.vx *= f;
173:   p.vz *= f;
174:   return sp;
175: }

177: // ===== Barrel rolling =====
178: // Barrels lie on their sides like logs, so a barrel only rolls BROADSIDE: the
179: // hit that knocks one into the field is the car measured perpendicular to its
180: // long axis. Mutates a plain {x,z,vx,vz,roll,axis,radius}; `axis` is the yaw of
181: // the long axis (world direction (cos axis, 0, -sin axis)) and the roll rate is
182: // the velocity component along the perpendicular (sin axis, 0, cos axis), over
183: // the radius â€” a no-slip wheel (v = Ï‰Â·r). Any along-axis part just skids.
184: // Returns the pre-friction speed so the caller knows whether it is still moving.
185: export function stepBarrel(b, dt, drag = 1.6) {
186:   const sp = Math.hypot(b.vx, b.vz);
187:   if (sp < 0.02) { b.vx = 0; b.vz = 0; return 0; }
188:   b.x += b.vx * dt;
189:   b.z += b.vz * dt;
190:   const perp = b.vx * Math.sin(b.axis) + b.vz * Math.cos(b.axis);
191:   b.roll += (perp / Math.max(0.2, b.radius || 0.5)) * dt;
192:   const f = Math.max(0, 1 - drag * dt);
193:   b.vx *= f;
194:   b.vz *= f;
195:   return sp;
196: }

203: // Tree positions on a staggered grid inside `rect`.
204: export function orchardGrid(rect = ORCHARD, spacing = APPLE_TREE.spacing) {
205:   const pts = [];
206:   const nx = Math.max(1, Math.floor((rect.x1 - rect.x0) / spacing));
207:   const nz = Math.max(1, Math.floor((rect.z1 - rect.z0) / spacing));
208:   const ox = (rect.x0 + rect.x1) / 2 - ((nx - 1) * spacing) / 2;
209:   const oz = (rect.z0 + rect.z1) / 2 - ((nz - 1) * spacing) / 2;
210:   for (let i = 0; i < nx; i++) {
211:     for (let j = 0; j < nz; j++) {
212:       const stagger = (j % 2) * spacing * 0.5;
213:       pts.push({ x: ox + i * spacing + stagger, z: oz + j * spacing });
214:     }
215:   }
216:   return pts;
217: }
